
module mara.font

use mara.core
use mara.string
use mara.math
use mara.endian
use mara.os
use mara.memory

Glyph :: struct {
	x, y, w, h: i32
	bearing_x, bearing_y, advance :f32
}

Atlas :: struct (bytes: []byte, u_per_em: u16) {
	size := 1024
	cell := 48
	pixels := 32 // used to scale em to pixels
	pad : f32 = 4
	per_row : i32 = size / cell
	px_scale := f32(pixels) / f32(u_per_em)
	data := bytes
}

load_glyphs :: fun(glyphs: ^[]Glyph, first: u32, font: cstr) -> [3 * MB]byte {
	ttf_file : [..2 * MB]byte
	read_err := read_entire_file(font, &ttf_file)
	if bool(read_err) do print(read_err)
	ttf := TTF(&ttf_file)

	storage : [3 * MB]byte
	atlas := Atlas(storage[:], ttf.head.units_per_em)

	for i in 0..i32(glyphs.len) {
		codepoint := first + u32(i)

		glyph_index := cmap_glyph_index(&ttf_file, ttf.cmap_sub, codepoint)
		goff := ttf.loca.offsets[glyph_index]
		glen := ttf.loca.offsets[glyph_index + 1] - goff
		gstart := ttf.glyf_table_loc + goff
		outline := parse_glyph(ttf_file[gstart : gstart + glen])

		glyphs[i].x = (i % atlas.per_row) * atlas.cell
		glyphs[i].y = (i / atlas.per_row) * atlas.cell
		// hmtx holds full {advance, lsb} records only for the first
		// num_of_long_hor_metrics glyphs; everything past that shares the
		// last record's advance (the array degrades to 2-byte lsb entries).
		metric := min(i64(glyph_index), i64(ttf.hhea.num_of_long_hor_metrics) - 1)
		adv_units := f32(read_u16be(ttf_file[ttf.hmtx_loc + i32(metric) * 4]))
		glyphs[i].advance = adv_units * atlas.px_scale

		rasterize_msdf(&outline, &glyphs[i], &atlas)
	}

	return storage
}

// ============================================================================
// Glyph → MSDF atlas
// ============================================================================
// Reconstruct a glyph's contours from parse_glyph's points (synthesizing the
// TrueType implied on-curve midpoints) and flatten the quadratics to line
// segments. Color the edges across the 3 channels (switching at corners), then
// per pixel take the min distance per channel with a shared winding sign and
// write the normalized signed distances as RGB into the atlas. The median of
// the three channels reconstructs the glyph, with corners kept sharp.

RASTER_QUAD_STEPS :: 6
MAX_RASTER_EDGES  :: 4096
MSDF_PX_RANGE     :: 4.0   // signed-distance range (px) mapped onto 0..1 / 0..255

Raster_Edge :: struct {
	x0 : f32
	y0 : f32
	x1 : f32
	y1 : f32
	color : i32   // channel mask: 1=R, 2=G, 4=B; 7=all (smooth)
}

// Rotate among the three two-channel edge colors (YELLOW=R+G=3, MAGENTA=R+B=5,
// CYAN=G+B=6). Any two share exactly one channel, so switching at a corner
// makes that corner the channels' crossing — which the median preserves.
next_edge_color :: fun(c: i32) -> i32 {
	if c == 3 { return 5 }
	if c == 5 { return 6 }
	return 3
}

// Corner: the two segments at a shared point turn sharply (cos(angle) below
// the threshold ≈ a >45° turn). Flattened curve steps stay well above it.
edge_is_corner :: fun(a: Raster_Edge, b: Raster_Edge) -> bool {
	dax := a.x1 - a.x0
	day := a.y1 - a.y0
	dbx := b.x1 - b.x0
	dby := b.y1 - b.y0
	la := sqrt(dax*dax + day*day)
	lb := sqrt(dbx*dbx + dby*dby)
	if la <= 0.0 or lb <= 0.0 { return false }
	cosang := (dax*dbx + day*dby) / (la * lb)
	return cosang < 0.7
}

// Rasterize the glyph's MSDF into the atlas. The caller has already placed the
// glyph: `glyph.x/y` are the cell's top-left pixel origin and `glyph.advance` is
// set, so this fills the rest — `glyph.w/h` (cell size in px) and
// `glyph.bearing_x/y` (pen→cell offsets) — then writes the cell's RGB distance
// field into `atlas.data`. The median of the three channels reconstructs the
// glyph with corners kept sharp.
rasterize_msdf :: fun(outline: ^Outline, glyph: ^Glyph, atlas: ^Atlas)
{
	if outline.num_contours <= 0 { glyph.w = 0 glyph.h = 0 return }

	scale := atlas.px_scale
	pad   := atlas.pad

	xmin := f32(outline.x_min)
	xmax := f32(outline.x_max)
	ymin := f32(outline.y_min)
	ymax := f32(outline.y_max)

	x_origin  := xmin - pad / scale
	y_origin  := ymax + pad / scale

	glyph.w = i32((xmax - xmin) * scale + 2.0 * pad) + 1
	glyph.h = i32((ymax - ymin) * scale + 2.0 * pad) + 1
	// The fill loop below writes w*h pixels at the cell origin — a glyph
	// bigger than its cell would bleed into the neighbor's pixels.
	assert(glyph.w <= atlas.cell)
	assert(glyph.h <= atlas.cell)
	glyph.bearing_x = xmin * scale - pad
	glyph.bearing_y = ymax * scale + pad

	edges : [..MAX_RASTER_EDGES]Raster_Edge
	contour_start : [..512]i64

	// similar to glyph point, but we won't merge them
	Node :: struct {
		x, y: f32
		on_curve : bool
	}
	nodes : [..1024]Node

	point_start := 0
	for ci in 0..outline.num_contours {
		point_end := outline.end_pts[ci]
		n_points := point_end - point_start + 1
		if n_points >= 2 {
			first_edge := edges.len
			// first on-curve point in the contour (relative index)
			first_on_curve := -1
			for k in 0..n_points {
				if outline.points[point_start + k].on_curve {
					first_on_curve = k
					break
				}
			}
			if first_on_curve < 0 { first_on_curve = 0 }

			// Build the node ring starting on-curve, inserting the implied
			// on-curve midpoints between consecutive off-curve control points.
			nodes.len = 0
			first_x := (f32(i32(outline.points[point_start + first_on_curve].x)) - x_origin) * scale
			first_y := (y_origin - f32(i32(outline.points[point_start + first_on_curve].y))) * scale
			nodes[nodes.len] = Node{first_x, first_y, true}
			nodes.len += 1
			have_off := false
			off_x : f32 = 0.0
			off_y : f32 = 0.0
			for j in 1..n_points {
				ring_idx := first_on_curve + j
				if ring_idx >= n_points { ring_idx = ring_idx - n_points }
				point := outline.points[point_start + ring_idx]
				point_x := (f32(i32(point.x)) - x_origin) * scale
				point_y := (y_origin - f32(i32(point.y))) * scale
				if point.on_curve {
					if have_off {
						nodes[nodes.len] = Node{off_x, off_y, false}
						nodes.len += 1
						have_off = false
					}
					nodes[nodes.len] = Node{point_x, point_y, true}
					nodes.len += 1
				elif have_off
					// two consecutive off-curve controls — TrueType implies an
					// on-curve point at their midpoint
					nodes[nodes.len] = Node{off_x, off_y, false} nodes.len += 1
					nodes[nodes.len] = Node{(off_x + point_x) * 0.5, (off_y + point_y) * 0.5, true} nodes.len += 1
					off_x = point_x off_y = point_y have_off = true
				else
					off_x = point_x off_y = point_y have_off = true
				}
			}
			// a contour can end on a trailing off-curve control
			if have_off { nodes[nodes.len] = Node{off_x, off_y, false} nodes.len += 1 }

			// Walk the ring, emitting flattened edges; close back to node 0.
			pen_x := nodes[0].x
			pen_y := nodes[0].y
			node_i : i32 = 1
			for node_i < nodes.len {
				if nodes[node_i].on_curve {
					if edges.len < MAX_RASTER_EDGES { edges[edges.len] = Raster_Edge{pen_x, pen_y, nodes[node_i].x, nodes[node_i].y, 0} edges.len += 1 }
					pen_x = nodes[node_i].x pen_y = nodes[node_i].y
					node_i += 1
				else
					ctrl_x := nodes[node_i].x
					ctrl_y := nodes[node_i].y
					end_x : f32 = 0.0
					end_y : f32 = 0.0
					next_i := node_i + 1
					if next_i >= nodes.len { end_x = nodes[0].x end_y = nodes[0].y else end_x = nodes[next_i].x end_y = nodes[next_i].y }
					prev_x := pen_x
					prev_y := pen_y
					for step in 0..RASTER_QUAD_STEPS {
						t := f32(step + 1) / f32(RASTER_QUAD_STEPS)
						inv_t := 1.0 - t
						curve_x := inv_t*inv_t*pen_x + 2.0*inv_t*t*ctrl_x + t*t*end_x
						curve_y := inv_t*inv_t*pen_y + 2.0*inv_t*t*ctrl_y + t*t*end_y
						if edges.len < MAX_RASTER_EDGES { edges[edges.len] = Raster_Edge{prev_x, prev_y, curve_x, curve_y, 0} edges.len += 1 }
						prev_x = curve_x prev_y = curve_y
					}
					pen_x = end_x pen_y = end_y
					node_i += 2
				}
			}
			if edges.len < MAX_RASTER_EDGES {
				edges[edges.len] = Raster_Edge{pen_x, pen_y, nodes[0].x, nodes[0].y, 0}
				edges.len += 1
			}
			if edges.len > first_edge {
				contour_start[contour_start.len] = first_edge
				contour_start.len += 1
			}
		}
		point_start = point_end + 1
	}
	// Sentinel: cap the last contour's edge range and count it too, so
	// contour_start holds (contours + 1) fenceposts — the coloring loop
	// reads contour ci as the span [contour_start[ci], contour_start[ci+1]).
	contour_start[contour_start.len] = i32(edges.len)
	contour_start.len += 1

	// Edge coloring: per contour, switch color at corners so adjacent spans
	// share exactly one channel (the median then keeps the corner sharp).
	// Smooth / single-corner contours stay white (all channels = plain SDF).
	is_corner : [..MAX_RASTER_EDGES]bool
	is_corner.len = MAX_RASTER_EDGES
	for ci in 0..(i32(contour_start.len) - 1) {
		edge_lo := contour_start[ci]
		edge_hi := contour_start[ci + 1]
		n_edges := edge_hi - edge_lo
		if n_edges >= 1 {
			corner_total := 0
			for ei in 0..n_edges {
				prev_ei := ei - 1
				if prev_ei < 0 { prev_ei = n_edges - 1 }
				if edge_is_corner(edges[edge_lo + prev_ei], edges[edge_lo + ei]) {
					is_corner[ei] = true
					corner_total += 1
				else
					is_corner[ei] = false
				}
			}
			if corner_total <= 1 {
				for ei in 0..n_edges { edges[edge_lo + ei].color = 7 }
			else
				first_corner := -1
				for ei in 0..n_edges {
					if is_corner[ei] {
						if first_corner < 0 { first_corner = ei }
					}
				}
				color := 3
				for step in 0..n_edges {
					ei := first_corner + step
					if ei >= n_edges { ei = ei - n_edges }
					if is_corner[ei] { color = next_edge_color(color) }
					edges[edge_lo + ei].color = color
				}
			}
		}
	}

	// Fill: per pixel, the signed distance — magnitude is the min distance to
	// any edge, sign from the non-zero winding rule (positive inside). One
	// sample per pixel; the field is smooth, so no supersampling.
	for py in 0..glyph.h {
		for px in 0..glyph.w {
			sample_x := f32(px) + 0.5
			sample_y := f32(py) + 0.5
			min_d2_r : f32 = 1000000.0
			min_d2_g : f32 = 1000000.0
			min_d2_b : f32 = 1000000.0
			winding : i32 = 0
			for ei in 0..i32(edges.len) {
				x0 := edges[ei].x0
				y0 := edges[ei].y0
				x1 := edges[ei].x1
				y1 := edges[ei].y1
				// squared distance from the sample to segment (x0,y0)→(x1,y1)
				dx := x1 - x0
				dy := y1 - y0
				len2 := dx*dx + dy*dy
				t : f32 = 0.0
				if len2 > 0.0 {
					t = max(0.0, min(1.0, ((sample_x - x0)*dx + (sample_y - y0)*dy) / len2))
				}
				proj_x := x0 + t*dx
				proj_y := y0 + t*dy
				dist_x := sample_x - proj_x
				dist_y := sample_y - proj_y
				d2 := dist_x*dist_x + dist_y*dist_y
				// this edge contributes to the channels it is colored for
				color := edges[ei].color
				if (color & 1) != 0 and d2 < min_d2_r { min_d2_r = d2 }
				if (color & 2) != 0 and d2 < min_d2_g { min_d2_g = d2 }
				if (color & 4) != 0 and d2 < min_d2_b { min_d2_b = d2 }
				// winding contribution (shared sign): the edge crosses this
				// scanline when exactly one endpoint is at/above it
				crosses := (y0 <= sample_y and y1 > sample_y) or (y0 > sample_y and y1 <= sample_y)
				if crosses {
					cross_t := (sample_y - y0) / (y1 - y0)
					x_crossing := x0 + cross_t * (x1 - x0)
					if sample_x < x_crossing {
						if y1 > y0 { winding += 1 else winding -= 1 }
					}
				}
			}
			// signed per-channel distances (positive inside) → normalized RGB
			dist_sign : f32 = -1.0
			if winding != 0 { dist_sign = 1.0 }
			out_r := max(0.0, min(1.0, 0.5 + (dist_sign * sqrt(min_d2_r)) / MSDF_PX_RANGE))
			out_g := max(0.0, min(1.0, 0.5 + (dist_sign * sqrt(min_d2_g)) / MSDF_PX_RANGE))
			out_b := max(0.0, min(1.0, 0.5 + (dist_sign * sqrt(min_d2_b)) / MSDF_PX_RANGE))
			dst_idx := ((glyph.y + py) * atlas.size + (glyph.x + px)) * 3
			atlas.data[dst_idx + 0] = u8(out_r * 255.0)
			atlas.data[dst_idx + 1] = u8(out_g * 255.0)
			atlas.data[dst_idx + 2] = u8(out_b * 255.0)
		}
	}
}

// Verify helper: read an atlas cell's RGB back, un-normalize, and ASCII-dump
// the median — confirms both the channel math and the buffer write/packing.
dump_atlas_cell :: fun(atlas: ^[]byte, atlas_w: i32, ax: i32, ay: i32, cw: i32, ch: i32) {
	for py in 0..ch {
		row : str
		for pxi in 0..cw {
			idx := ((ay + py) * atlas_w + (ax + pxi)) * 3
			dr := (f32(i32(atlas[idx + 0])) / 255.0 - 0.5) * MSDF_PX_RANGE
			dg := (f32(i32(atlas[idx + 1])) / 255.0 - 0.5) * MSDF_PX_RANGE
			db := (f32(i32(atlas[idx + 2])) / 255.0 - 0.5) * MSDF_PX_RANGE
			med := max(min(dr, dg), min(max(dr, dg), db))
			if med > 1.5 {
				&row + "#"
			elif med > 0.0
				&row + "+"
			elif med > -1.5
				&row + "."
			else
				&row + " "
			}
		}
		print(row)
	}
}

// Spec type aliases. Transparent aliases over the underlying integer width —
// arithmetic on them behaves like the underlying int. The fixed-point ones
// need explicit conversion to f32 to get their real-number value.
Fixed          :: type i32  // 16.16 signed fixed-point
F2_Dot_14      :: type i16  // 2.14 signed fixed-point
Short_Frac     :: type i16  // 1.15 signed fraction
F_Word         :: type i16  // signed integer in font design units (FUnits)
U_F_Word       :: type u16  // unsigned integer in FUnits
Long_Date_Time :: type i64  // seconds since 1904-01-01 00:00

// The parsed font, shaped after the on-disk file: the spec types nested (reach
// them as TTF.Head etc.) and the tables we read, filled into fields by the
// constructor's parse body at the bottom. load_glyphs builds one of these and
// bakes from it — it parses nothing of its own.
TTF :: struct (bytes: ^[]byte) {

	Offset_Table :: struct {
		scaler_type : u32
		num_tables : u16
		search_range : u16
		entry_selector: u16
		range_shift : u16
		tables: [..32]Table
	}

	Table :: struct {
		tag: [4]utf8
		check_sum : u32
		offset: u32
		len: u32
	}

	#packed Head :: struct {
		version              : Fixed          // 0x00010000 = version 1.0
		font_revision        : Fixed          // set by font manufacturer
		check_sum_adjustment : u32            // sum the font as u32, store 0xB1B0AFBA - sum
		magic_number         : u32            // must be 0x5F0F3CF5
		flags                : u16            // bit 0 baseline at y=0, 1 lsb at x=0, 7 needs layout, 9 contains RTL, ...
		units_per_em         : u16            // 64..16384
		created              : Long_Date_Time
		modified             : Long_Date_Time
		x_min                : F_Word         // min over all glyph bboxes
		y_min                : F_Word
		x_max                : F_Word
		y_max                : F_Word
		mac_style            : u16            // bit 0 bold, 1 italic, 2 underline, 3 outline, 4 shadow, 5 condensed, 6 extended
		lowest_rec_ppem      : u16            // smallest readable size in pixels
		font_direction_hint  : i16            // 0 mixed, 1 LTR, 2 LTR+neutrals, -1 RTL, -2 RTL+neutrals
		index_to_loc_format  : i16            // 0 = short (u16) offsets in loca, 1 = long (u32)
		glyph_data_format    : i16            // 0 for current format
	}

	Maxp :: struct {
		version                  : Fixed // 0x00010000 (TrueType v1.0) or 0x00005000 (CFF v0.5)
		num_glyphs               : u16   // total glyph count in the font
		max_points               : u16   // max points in a non-composite glyph
		max_contours             : u16   // max contours in a non-composite glyph
		max_composite_points     : u16   // max points in a composite glyph
		max_composite_contours   : u16   // max contours in a composite glyph
		max_zones                : u16   // bytecode hinting; 1 if no twilight zone, else 2
		max_twilight_points      : u16   // bytecode hinting; max twilight zone points
		max_storage              : u16   // bytecode hinting; max storage area locations
		max_function_defs        : u16   // bytecode hinting; max FDEFs
		max_instruction_defs     : u16   // bytecode hinting; max IDEFs
		max_stack_elements       : u16   // bytecode hinting; max stack depth
		max_size_of_instructions : u16   // bytecode hinting; max byte count for any glyph's instructions
		max_component_elements   : u16   // max top-level components per composite glyph
		max_component_depth      : u16   // max recursion depth of composite glyphs
	}

	Cmap :: struct {
		version       : u16
		num_subtables : u16
		encodings     : [..16]Encoding_Record
	}

	Encoding_Record :: struct {
		platform_id          : u16 // 0 Unicode, 1 Mac, 3 Microsoft
		platform_specific_id : u16
		offset               : u32 // byte offset from start of cmap table to this encoding's subtable
	}

	Loca :: struct {
		// we read this with a function rather than directly
		// because the data here can be either u16 or u32
		offsets : [..1 + (1<<16)]u32
	}

	Glyf :: struct {
		data : []Glyph // has to be decoded from raw bytes
	}

	Glyph :: struct {
		num_contours : i16   // < 0 means composite

		x_min : i16
		y_min : i16
		x_max : i16
		y_max : i16
		// the first 5 fields can be byte read

		data : Simple // can't be read, must be decoded

		// GET A SECOND STRUCT THAT WILL STORE DECODED DATA

		Simple :: struct {
			contour_end_pts : []u16 // len is num_contours
			instructions_len: u16
			instructions: []u8
			flags : []u8
			rel_x : []byte // each x is relative to previous
			rel_y : []byte // each y is relative to previous
			// first elem relative to 0 in both arrays
		}
		// skipping compound glyphs for now
	}


	Hhea :: struct {
		version                 : Fixed     // 0x00010000
		ascent                  : F_Word    // typographic ascent
		descent                 : F_Word    // typographic descent (negative)
		line_gap                : F_Word    // extra recommended leading
		advance_width_max       : U_F_Word
		min_left_side_bearing   : F_Word
		min_right_side_bearing  : F_Word
		x_max_extent            : F_Word
		caret_slope_rise        : i16       // 1 for vertical caret
		caret_slope_run         : i16       // 0 for vertical caret
		caret_offset            : F_Word    // 0 for non-slanted fonts
		reserved1               : i16       // must be 0
		reserved2               : i16       // must be 0
		reserved3               : i16       // must be 0
		reserved4               : i16       // must be 0
		metric_data_format      : i16       // 0 for current format
		num_of_long_hor_metrics : u16       // count of full metrics records in hmtx; trailing glyphs share advance
	}

	Hmtx :: struct {
		// hhea.num_of_long_hor_metrics
		long_metrics  : []Long_Hor_Metric
		// num_glyphs - num_of_long_hor_metrics 
		side_bearings : []F_Word
	}

	Long_Hor_Metric :: struct {
		advance_width     : U_F_Word
		left_side_bearing : F_Word
	}

	// ─── parse: read the table directory, then the tables we use, into fields ───
	h : Offset_Table = #big_endian bytes[0]
	h.tables = #big_endian bytes[12]
	h.tables.len = h.num_tables

	head_loc := h.tables.get_offset("head")
	maxp_loc := h.tables.get_offset("maxp")
	cmap_loc := h.tables.get_offset("cmap")
	loca_loc := h.tables.get_offset("loca")
	glyf_table_loc := h.tables.get_offset("glyf")
	hhea_loc := h.tables.get_offset("hhea")
	hmtx_loc := h.tables.get_offset("hmtx")

	head : Head = #big_endian bytes[head_loc]
	maxp : Maxp = #big_endian bytes[maxp_loc]
	cmap : Cmap = #big_endian bytes[cmap_loc]
	loca : Loca = loca_correct_format(head, maxp, bytes[loca_loc:])
	hhea : Hhea = #big_endian bytes[hhea_loc]
	// hmtx : Hmtx = hmtx_prepare(hhea, maxp, bytes[hmtx_loc:])

	// cmap encoding records sit right after the 4-byte header; fill them, then
	// pick a Unicode subtable to map character codes through.
	cmap.encodings = #big_endian bytes[cmap_loc + 4]
	cmap.encodings.len = i32(cmap.num_subtables)
	cmap_sub := pick_cmap_subtable(cmap, cmap_loc)

	get_offset :: fun(tables: [..32]Table, target: []utf8) -> i32 {
		for table in tables {
			if starts_with(table.tag[:], target) {
				return i32(table.offset)
			}
		}
		print("Table '%' not found.", target)
		return 0
	}

	loca_correct_format :: fun(head: Head, maxp: Maxp,
		bytes: []byte) -> loca: Loca 
	{
		loca.offsets.len = i32(maxp.num_glyphs + 1)		
		if head.index_to_loc_format == 0 {
			off16 : [..1 + (1<<16)]u16 = #big_endian bytes[0]
			off16.len = loca.offsets.len
			for item, i in off16 {
				loca.offsets[i] = u32(item) * 2
			}
		else
			loca.offsets = #big_endian bytes[0]
		}
		return loca
	}

	// Choose which cmap subtable to map characters through, returning its
	// absolute file offset (0 if none). Prefer Microsoft Unicode BMP (3,1) —
	// the format-4 table nearly every font ships — then any Unicode-platform
	// (0,*) table, then whatever's first as a fallback.
	pick_cmap_subtable :: fun(cmap: Cmap, cmap_loc: i32) -> i32 {
		best : i32 = -1
		best_score : i32 = -1
		for e, i in cmap.encodings {
			score : i32 = 1
			if e.platform_id == 0 { score = 2 }
			if e.platform_id == 3 and e.platform_specific_id == 1 { score = 3 }
			if score > best_score {
				best_score = score
				best = i
			}
		}
		if best < 0 { return 0 }
		return cmap_loc + i32(cmap.encodings[best].offset)
	}
}

// ============================================================================
// Character map (cmap)
// ============================================================================
//
// Maps a character code to a glyph index. The cmap table holds one or more
// encoding subtables; the TTF constructor (pick_cmap_subtable) chose one,
// preferring a Unicode mapping, and passes its absolute file offset here.
// Only format 4 (segment-mapped BMP — the near-universal desktop format) and
// format 0 (legacy 256-byte) are decoded; anything else hits a diagnostic
// rather than silently returning .notdef (glyph 0).

cmap_glyph_index :: fun(bytes: ^[]byte, subtable_start: i32, char: u32) -> u16 {
	if subtable_start == 0 { return 0 }
	format := read_u16be(bytes[subtable_start])
	if format == 4 {
		return cmap_format4_lookup(bytes, subtable_start, char)
	elif format == 0
		return cmap_format0_lookup(bytes, subtable_start, char)
	else
		print("cmap: unsupported subtable format %", i64(format))
		return 0
	}
}

// Format 4: four parallel u16 arrays of seg_count entries — end codes, start
// codes, idDelta, idRangeOffset — then the glyphIdArray. Subtable layout:
// format(2) length(2) language(2) segCountX2(2) searchRange(2)
// entrySelector(2) rangeShift(2), so the arrays begin at +14 with a 2-byte
// reservedPad between the end-code and start-code arrays.
cmap_format4_lookup :: fun(bytes: ^[]byte, sub: i32, char: u32) -> u16 {
	seg_x2 := i32(read_u16be(bytes[sub + 6]))
	end_codes   := sub + 14
	start_codes := end_codes + seg_x2 + 2     // +2 skips reservedPad
	id_deltas   := start_codes + seg_x2
	id_ranges   := id_deltas + seg_x2
	seg_count := seg_x2 / 2
	for s in 0..seg_count {
		end_code := u32(read_u16be(bytes[end_codes + s * 2]))
		if char <= end_code {
			start_code := u32(read_u16be(bytes[start_codes + s * 2]))
			if char < start_code { return 0 }
			id_delta := read_u16be(bytes[id_deltas + s * 2])
			id_range := i32(read_u16be(bytes[id_ranges + s * 2]))
			if id_range == 0 {
				// Direct: (char + idDelta) mod 65536. u16() truncates the wide
				// u32 sum to its low 16 bits, which is the modular reduction
				// the spec wants (idDelta is a u16-encoded signed value).
				return u16(char + u32(id_delta))
			else
				// Indirect: glyphIdArray address, relative to this
				// idRangeOffset entry's own location.
				addr := id_ranges + s * 2 + id_range + 2 * (i32(char) - i32(start_code))
				g := read_u16be(bytes[addr])
				if g == 0 { return 0 }
				return u16(u32(g) + u32(id_delta))
			}
		}
	}
	return 0
}

// Format 0: a flat 256-entry byte map at +6 (after format/length/language).
cmap_format0_lookup :: fun(bytes: ^[]byte, sub: i32, char: u32) -> u16 {
	if char > 255 { return 0 }
	return u16(bytes[sub + 6 + i32(char)])
}

// ============================================================================
// Glyph parsing
// ============================================================================
//
// A parsed glyph lives outside the file-format Layout struct — it's the
// usable form for rendering. parse_glyph reads from the raw glyf-table bytes
// and writes into a caller-allocated Glyph. The Glyph's partial arrays bound
// per-glyph contour/point counts statically; the per-glyph .len gets set to
// what the glyph actually contains.

MAX_GLYPH_POINTS   :: 4096
Glyph_Point :: struct {
	x        : i16
	y        : i16
	on_curve : bool
}

Outline :: struct {
	num_contours : i16   // < 0 means composite

	// Bounding box in FUnits, copied from the per-glyph header.
	x_min : i16
	y_min : i16
	x_max : i16
	y_max : i16

	// Simple-glyph: end_pts[i] is the LAST point index of contour i.
	// points is the flattened per-glyph point list across all contours.
	end_pts : [..256]u16
	points  : [..MAX_GLYPH_POINTS]Glyph_Point
}

parse_glyph :: fun(glyph_bytes: []byte) -> Outline {
	glyph : Outline = #big_endian glyph_bytes[:]
	if glyph_bytes.len == 0 do return glyph

	if glyph.num_contours < 0 {
		// Composite glyph — skip for now
		glyph.end_pts.len = 0
		glyph.points.len = 0
		return glyph
	else 
		// this function only does simple glyphs
		parse_simple_glyph_body(&glyph, glyph_bytes)
		return glyph
	}
}

// Simple-glyph flag bits. Bits 6-7 are reserved (must be 0 per spec).
GLYPH_FLAG_ON_CURVE :: u8(0x01)
GLYPH_FLAG_X_SHORT  :: u8(0x02)   // x delta is 1 byte unsigned
GLYPH_FLAG_Y_SHORT  :: u8(0x04)   // y delta is 1 byte unsigned
GLYPH_FLAG_REPEAT   :: u8(0x08)   // next byte = repeat count for this flag
GLYPH_FLAG_X_SAME   :: u8(0x10)   // if X_SHORT: dx is positive; else: same as prev
GLYPH_FLAG_Y_SAME   :: u8(0x20)   // analogous for y
parse_simple_glyph_body :: fun(glyph: ^Outline, bytes : []byte) {
	n_contours := i32(glyph.num_contours)
	// glyph end points start rght aftre the glyph header?
	// so we just add 10 to cursor and we're good to go?
	cursor := 10

	// End-point index per contour. Last entry + 1 is the total point count.
	glyph.end_pts.len = n_contours
	for i in 0..n_contours {
		glyph.end_pts[i] = read_u16be(bytes[cursor])
		cursor += 2
	}
	n_points := i32(glyph.end_pts[n_contours - 1]) + 1
	assert(n_points <= MAX_GLYPH_POINTS)
	glyph.points.len = n_points

	// TrueType hinting bytecode — the spec keeps it inline between the
	// endpoints and the flags. We don't interpret it; skip past.
	instr_len := i32(read_u16be(bytes[cursor]))
	cursor += 2
	cursor += instr_len

	// Per-point flag bytes, with REPEAT-byte run-length decoding: when bit 3
	// of a flag is set, the NEXT byte is a count of additional points that
	// share the same flag. Decoded into a flat flags-per-point local array
	// so the coord-decode loops below can index by point index cheaply.
	flags : [..MAX_GLYPH_POINTS]u8
	flags.len = n_points
	p : i32 = 0
	for p < n_points {
		f := u8(bytes[cursor]); cursor += 1
		flags[p] = f
		p += 1
		if (f & GLYPH_FLAG_REPEAT) != 0 {
			repeat := i32(bytes[cursor]); cursor += 1
			for r in 0..repeat {
				flags[p] = f
				p += 1
			}
		}
	}

	// X coordinates: each point's x is a signed delta from the previous
	// point. Encoding per point depends on X_SHORT (1-byte vs 2-byte) and
	// X_SAME (sign-bit-for-short, or same-as-prev-for-long).
	x : i32 = 0
	for i in 0..n_points {
		f := flags[i]
		if (f & GLYPH_FLAG_X_SHORT) != 0 {
			dx := i32(bytes[cursor]); cursor += 1
			if (f & GLYPH_FLAG_X_SAME) == 0 { dx = -dx }
			x += dx
		elif (f & GLYPH_FLAG_X_SAME) == 0
			x += i32(read_i16be(bytes[cursor])); cursor += 2
		}
		// X_SAME && !X_SHORT: no bytes — delta is 0, x unchanged.
		glyph.points[i].x = i16(x)
		glyph.points[i].on_curve = (f & GLYPH_FLAG_ON_CURVE) != 0
	}

	// Y coordinates: same shape with Y_SHORT/Y_SAME bits.
	y : i32 = 0
	for i in 0..n_points {
		f := flags[i]
		if (f & GLYPH_FLAG_Y_SHORT) != 0 {
			dy := i32(bytes[cursor]); cursor += 1
			if (f & GLYPH_FLAG_Y_SAME) == 0 { dy = -dy }
			y += dy
		elif (f & GLYPH_FLAG_Y_SAME) == 0
			y += i32(read_i16be(bytes[cursor])); cursor += 2
		}
		glyph.points[i].y = i16(y)
	}
	// A well-formed glyph consumes exactly its slice; past the end means the
	// flag/coord decode desynced (bounds checks catch reads, not the drift).
	assert(cursor <= bytes.len)
}
